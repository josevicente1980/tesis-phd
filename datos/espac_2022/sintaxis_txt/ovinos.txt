  /VLABELS VARIABLES=gv_tn gv_trs gv_trc gv_tro  ual_prov DISPLAY=NONE  /VLABELS 
    VARIABLES=gv_k1001 DISPLAY=LABEL
  /TABLE gv_tn + gv_trs + gv_trc + gv_tro + ual_prov BY gv_k1001 [SUM]
  /SLABELS VISIBLE=NO
  /CATEGORIES VARIABLES=gv_tn gv_trs gv_trc gv_tro ual_prov ORDER=A KEY=VALUE EMPTY=INCLUDE.

USE ALL. 
EXECUTE.
